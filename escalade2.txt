        function escapeHtml(str) {
            const div = document.createElement("div");
            div.textContent = String(str);
            return div.innerHTML;
        }

        function formatPoints(num) {
            if (num === null || num === undefined || isNaN(num)) return "0";
            return Number.isInteger(num) ? String(num) : String(num).replace(".", ",");
        }

        function getClimberDisplayName(index) {
            const raw = (appState.climbers[index] || "").trim();
            return raw ? raw : `Grimpeur ${index + 1}`;
        }

        function getRouteDisplayLabel(routeNumber) {
            return routeNumber ? `Voie ${routeNumber}` : "Non choisie";
        }

        function getClimberRoutePoints(routeIdx, climberIdx) {
            const routeObj = appState.routes[routeIdx];
            const height = routeObj.heights[climberIdx];
            if (!routeObj.selectedRoute || !height || !ROUTES_SCALE[routeObj.selectedRoute]) {
                return 0;
            }
            return ROUTES_SCALE[routeObj.selectedRoute][height] || 0;
        }

        function getRouteClimbersSum(routeIdx) {
            let sum = 0;
            for (let c = 0; c < appState.groupSize; c++) {
                sum += getClimberRoutePoints(routeIdx, c);
            }
            return sum;
        }

        function getRouteTotalPoints(routeIdx) {
            const baseSum = getRouteClimbersSum(routeIdx);
            return appState.routes[routeIdx].allTopBonus === "OUI" ? baseSum + 10 : baseSum;

                    <td colspan="2" style="color:#0c4a6e; font-size:0.98rem;">
                        ${formatPoints(t3)} pt${t3 > 1 ? "s" : ""}
                        ${appState.routes[2].allTopBonus === "OUI" ? '<div style="font-size:0.7rem; color:#059669;">(dont +10 bonus TOP)</div>' : ""}

                </tr>
            `);

        function updateCompletionIndicator() {
            const missingParts = [];

            let missingNames = 0;
            for (let c = 0; c < appState.groupSize; c++) {
                if (!appState.climbers[c].trim()) missingNames++;
            }
            if (missingNames > 0) missingParts.push(`${missingNames} prénom${missingNames > 1 ? "s" : ""}`);

            let missingRoutes = 0;
            for (let r = 0; r < 3; r++) {
                if (!appState.routes[r].selectedRoute) missingRoutes++;
            }
            if (missingRoutes > 0) missingParts.push(`${missingRoutes} voie${missingRoutes > 1 ? "s" : ""} non choisie${missingRoutes > 1 ? "s" : ""}`);

            let missingHeights = 0;
            for (let r = 0; r < 3; r++) {
                for (let c = 0; c < appState.groupSize; c++) {
                    if (!appState.routes[r].heights[c]) missingHeights++;
                }
            }
            if (missingHeights > 0) missingParts.push(`${missingHeights} hauteur${missingHeights > 1 ? "s" : ""}`);

            const badge = document.getElementById("completionStatusBadge");
            const notice = document.getElementById("incompleteNotice");
            const missingText = document.getElementById("missingFieldsText");
            const sendBanner = document.getElementById("sendBannerCard");

            if (missingParts.length === 0) {
                badge.textContent = "Saisie complète";
                badge.classList.add("complete");
                notice.classList.add("hidden");
                sendBanner.classList.add("ready-complete");
            } else {
                badge.textContent = "Saisie partielle (envoi possible)";
                badge.classList.remove("complete");
                notice.classList.remove("hidden");
                sendBanner.classList.remove("ready-complete");
                missingText.textContent = `Éléments manquants (${missingParts.join(", ")}) : ils seront mentionnés comme « Non renseigné » (0 pt) dans le mail.`;
            }
        }

        /* =========================================================================
           7. FORMATAGE DU MAIL, COPIE & OUVERTURE MODALE D'ENVOI
           ========================================================================= */
        function buildFormattedEmailData() {
            const t1 = getRouteTotalPoints(0);
            const t2 = getRouteTotalPoints(1);
            const t3 = getRouteTotalPoints(2);
            const penalties = appState.safetyFaults * 3;
            const sessionTotal = (t1 + t2 + t3) - penalties;

            const v1Label = getRouteDisplayLabel(appState.routes[0].selectedRoute);
            const v2Label = getRouteDisplayLabel(appState.routes[1].selectedRoute);
            const v3Label = getRouteDisplayLabel(appState.routes[2].selectedRoute);

            const names = [];
            for (let c = 0; c < appState.groupSize; c++) {
                names.push(getClimberDisplayName(c));
            }

            const lines = [
                "RÉSULTATS DE LA SÉANCE D'ESCALADE",
                "==================================================",
                `Groupe (${appState.groupSize} personnes) : ${names.join(", ")}`,
                "",
                "DÉTAIL PAR GRIMPEUR :",
                "--------------------------------------------------"
            ];

            for (let c = 0; c < appState.groupSize; c++) {
                const name = getClimberDisplayName(c);

                const h1 = appState.routes[0].heights[c] || "Non renseigné";
                const p1 = (appState.routes[0].selectedRoute && appState.routes[0].heights[c])
                    ? `${formatPoints(getClimberRoutePoints(0, c))} pts`
                    : "0 pt";

                const h2 = appState.routes[1].heights[c] || "Non renseigné";
                const p2 = (appState.routes[1].selectedRoute && appState.routes[1].heights[c])
                    ? `${formatPoints(getClimberRoutePoints(1, c))} pts`
                    : "0 pt";

                const h3 = appState.routes[2].heights[c] || "Non renseigné";
                const p3 = (appState.routes[2].selectedRoute && appState.routes[2].heights[c])
                    ? `${formatPoints(getClimberRoutePoints(2, c))} pts`
                    : "0 pt";

                lines.push(`• ${name} :`);
                lines.push(`   - 1ère Voie (${v1Label}) : ${h1} -> ${p1}`);
                lines.push(`   - 2ème Voie (${v2Label}) : ${h2} -> ${p2}`);
                lines.push(`   - 3ème Voie (${v3Label}) : ${h3} -> ${p3}`);
            }

            lines.push("");
            lines.push("TOTAUX PAR VOIE :");
            lines.push("--------------------------------------------------");
            lines.push(`• Total 1ère Voie (${v1Label}) : ${formatPoints(t1)} pts (Tous au TOP : ${appState.routes[0].allTopBonus})`);
            lines.push(`• Total 2ème Voie (${v2Label}) : ${formatPoints(t2)} pts (Tous au TOP : ${appState.routes[1].allTopBonus})`);
            lines.push(`• Total 3ème Voie (${v3Label}) : ${formatPoints(t3)} pts (Tous au TOP : ${appState.routes[2].allTopBonus})`);
            lines.push("");
            lines.push("BILAN SÉCURITÉ & SCORE FINAL :");
            lines.push("==================================================");
            lines.push(`• Fautes sécurité : ${appState.safetyFaults} (Points pénalités : -${formatPoints(penalties)} pts)`);
            lines.push(`• NOMBRE TOTAL DE POINTS SUR LA SÉANCE : ${formatPoints(sessionTotal)} pts`);
            lines.push("==================================================");

            const subject = `Résultats Séance Escalade - Groupe (${names.join(", ")})`;
            const body = lines.join("\n");

            return {
                recipient: "romain.griot@gmail.com",
                subject,
                body,
                mailtoHref: `mailto:romain.griot@gmail.com?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`,
                gmailHref: `https://mail.google.com/mail/?view=cm&fs=1&to=romain.griot@gmail.com&su=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`
            };
        }

        function copyTextToClipboard(text, buttonLabelEl) {
            const textarea = document.createElement("textarea");
            textarea.value = text;
            textarea.style.position = "fixed";
            textarea.style.opacity = "0";
            document.body.appendChild(textarea);
            textarea.focus();
            textarea.select();

            let success = false;
            try {
                success = document.execCommand("copy");
            } catch (err) {
                success = false;
            }
            document.body.removeChild(textarea);

            if (success) {
                showToast("Récapitulatif copié dans le presse-papier !");
                if (buttonLabelEl) {
                    const originalText = buttonLabelEl.textContent;
                    buttonLabelEl.textContent = "Copié !";
                    setTimeout(() => {
                        buttonLabelEl.textContent = originalText;
                    }, 2200);
                }
            } else {
                showToast("Sélectionnez et copiez le texte affiché ci-dessous.");
            }
        }

        function openEmailModalAndTriggerMailto() {
            const emailData = buildFormattedEmailData();

            document.getElementById("mailtoAnchorBtn").setAttribute("href", emailData.mailtoHref);
            document.getElementById("gmailAnchorBtn").setAttribute("href", emailData.gmailHref);
            document.getElementById("modalEmailPreview").textContent = emailData.body;
            document.getElementById("emailModal").classList.remove("hidden");

            // Tentative d'ouverture automatique du client mail via un lien dynamique sans bloquer l'iframe
            try {
                const tempLink = document.createElement("a");
                tempLink.href = emailData.mailtoHref;
                tempLink.target = "_top";
                tempLink.style.display = "none";
                document.body.appendChild(tempLink);
                tempLink.click();
                document.body.removeChild(tempLink);
            } catch (e) {
                // La modale ouverte permet dans tous les cas un clic direct ou l'envoi via Gmail Web
            }
        }

        function attachRouteEventListeners() {
            // Sélection du numéro de voie (Voie 1 à Voie 13)
            document.querySelectorAll(".route-number-select").forEach(select => {
                select.addEventListener("change", (e) => {
                    const rIdx = parseInt(e.target.dataset.route, 10);
                    const val = e.target.value;
                    appState.routes[rIdx].selectedRoute = val ? parseInt(val, 10) : "";
                    e.target.classList.toggle("filled", Boolean(val));
                    updateAllCalculations();
                });
            });

            // Sélection de la hauteur atteinte par grimpeur
            document.querySelectorAll(".height-select").forEach(select => {
                select.addEventListener("change", (e) => {
                    const rIdx = parseInt(e.target.dataset.route, 10);
                    const cIdx = parseInt(e.target.dataset.climber, 10);
                    const val = e.target.value;
                    appState.routes[rIdx].heights[cIdx] = val;
                    e.target.classList.toggle("filled", Boolean(val));

                    // Vérification automatique : si tous les grimpeurs du groupe sont au TOP
                    let allAreTop = true;
                    for (let c = 0; c < appState.groupSize; c++) {
                        if (appState.routes[rIdx].heights[c] !== "TOP") {
                            allAreTop = false;
                            break;
                        }
                    }

                    const bonusSelect = document.getElementById(`allTopSelect-${rIdx}`);
                    if (allAreTop && appState.routes[rIdx].allTopBonus !== "OUI") {
                        appState.routes[rIdx].allTopBonus = "OUI";
                        if (bonusSelect) {
                            bonusSelect.value = "OUI";
                            bonusSelect.classList.add("filled");
                        }
                        showToast(`Bonus +10 pts activé (tous au TOP sur la ${ROUTE_META[rIdx].title}) !`);
                    } else if (!allAreTop && appState.routes[rIdx].allTopBonus === "OUI") {
                        appState.routes[rIdx].allTopBonus = "NON";
                        if (bonusSelect) {
                            bonusSelect.value = "NON";
                            bonusSelect.classList.remove("filled");
                        }
                    }

                    updateAllCalculations();
                });
            });

            // Sélection manuelle "Est-ce que tous les grimpeurs sont arrivés au TOP ? OUI/NON"
            document.querySelectorAll(".all-top-select").forEach(select => {
                select.addEventListener("change", (e) => {
                    const rIdx = parseInt(e.target.dataset.route, 10);
                    appState.routes[rIdx].allTopBonus = e.target.value;
                    e.target.classList.toggle("filled", e.target.value === "OUI");
                    updateAllCalculations();
                });
            });
        }
